async function runTool(name,a,allowWrites=true){addActivity(name,a?.path||a?.query||a?.prefix||'','tool');if(name==='repo_info')return {repository:S.repo.full_name,branch:S.branch,default_branch:S.repo.default_branch,file_count:visibleTree().length,staged:[...S.stage.keys()]};if(name==='list_files'){const p=a.prefix||'';return visibleTree().filter(x=>!p||x.path.startsWith(p)).slice(0,Math.min(250,a.limit||120)).map(x=>({path:x.path,size:x.size}))}if(name==='find_files'){const q=String(a.query||'').toLowerCase();return visibleTree().filter(x=>x.path.toLowerCase().includes(q)).slice(0,120).map(x=>({path:x.path,size:x.size}))}if(name==='read_file')return {path:a.path,content:await readFile(a.path)};if(name==='read_files'){const out=[];for(const p of (a.paths||[]).slice(0,6)){try{out.push({path:p,content:await readFile(p)})}catch(e){out.push({path:p,error:e.message})}}return out}if(!allowWrites&&name.startsWith('stage_'))return {error:'Read-only role'};if(name==='stage_file')return stageFile(a.path,a.content,a.reason||'');if(name==='stage_delete')return stageDelete(a.path,a.reason||'');return {error:'Unknown tool'}}

async function preAnalyzeAttachments(attachments){
  if(!attachments.length)return '';
  const summaries=[];
  for(const a of attachments){
    if(a.type==='file'&&a.text!=null){summaries.push(`FILE ${a.name}:\n${a.text.slice(0,50000)}`);continue}
    if(a.type==='file'&&a.mime==='application/pdf')continue;
    const modality=a.type==='image'?'image':'video';if(S.modelCaps[modality])continue;
    const helper=freeModelFor(modality);if(!helper){summaries.push(`[${a.name}: current model may not support ${modality}; media is attached but may be rejected.]`);continue}
    try{addActivity('Reading '+a.name,'via '+helper,'search');const content=[{type:'text',text:'Describe this attachment for another agent. Preserve visible text, UI details, errors, actions, and any business-relevant information. Be concise but specific.'}];if(modality==='image')content.push({type:'image_url',image_url:{url:a.data}});else content.push({type:'video_url',video_url:{url:a.data}});const r=await fetch('https://openrouter.ai/api/v1/chat/completions',{method:'POST',headers:{Authorization:'Bearer '+S.or,'Content-Type':'application/json'},body:JSON.stringify({model:helper,messages:[{role:'user',content}],temperature:.1,max_tokens:1800})});const d=await r.json();if(r.ok&&d.choices?.[0]?.message?.content)summaries.push(`[${a.name} analyzed by ${helper}]\n${d.choices[0].message.content}`)}catch(e){summaries.push(`[${a.name}: media pre-analysis failed: ${e.message}]`)}
  }
  return summaries.join('\n\n');
}
function userContent(task,attachments){const c=[{type:'text',text:task}];for(const a of attachments){if(a.type==='image'&&S.modelCaps.image)c.push({type:'image_url',image_url:{url:a.data}});if(a.type==='video'&&S.modelCaps.video)c.push({type:'video_url',video_url:{url:a.data}});if(a.type==='file'&&a.mime==='application/pdf'&&a.data)c.push({type:'file',file:{filename:a.name,file_data:a.data}});if(a.type==='file'&&a.text!=null)c[0].text+=`\n\n--- ATTACHED FILE: ${a.name} ---\n${a.text.slice(0,50000)}`}return c.length===1?c[0].text:c}
function providerChain(){const out=[{id:'or',label:'OpenRouter',key:S.or,url:'https://openrouter.ai/api/v1/chat/completions',model:S.model}];if(S.ocEnabled&&S.ocKey)out.push({id:'oc',label:'OpenCode',key:S.ocKey,url:'https://opencode.ai/zen/v1/chat/completions',model:S.ocModel});if(S.nousEnabled&&S.nousKey)out.push({id:'nous',label:'Nous',key:S.nousKey,url:'https://inference-api.nousresearch.com/v1/chat/completions',model:S.nousModel});if(S.providerSticky){const i=out.findIndex(x=>x.id===S.providerSticky);if(i>0){const [x]=out.splice(i,1);out.unshift(x)}}return out}
async function providerCall(p,messages,tools){const ctl=new AbortController();S.abort=ctl;const timer=setTimeout(()=>ctl.abort(),120000);try{const body={model:p.model,messages,temperature:.12,max_tokens:24000,tools,tool_choice:'auto'};if(S.web&&p.id==='or')body.tools=[...(tools||[]),{type:'openrouter:web_search',parameters:{max_results:4,max_total_results:8}},{type:'openrouter:web_fetch',parameters:{max_uses:4,max_content_tokens:12000}}];if(p.id==='nous')body.tags=['product=ox-mobile','user=phone'];const r=await fetch(p.url,{method:'POST',signal:ctl.signal,headers:{Authorization:'Bearer '+p.key,'Content-Type':'application/json',...(p.id==='or'?{'X-OpenRouter-Title':'OX Mobile'}:{})},body:JSON.stringify(body)});const raw=await r.text(),d=safeJSON(raw,{error:{message:raw}});if(!r.ok)throw Error(p.label+' '+r.status+': '+(d.error?.message||raw));if(!d.choices?.[0]?.message)throw Error(p.label+' returned no message');return d.choices[0].message}finally{clearTimeout(timer);if(S.abort===ctl)S.abort=null}}
async function routed(messages,tools){const errs=[];for(const p of providerChain())try{if(errs.length)addActivity('Fallback','Switching to '+p.label,'error');const m=await providerCall(p,messages,tools);S.providerSticky=p.id;return m}catch(e){errs.push(e.message);addActivity('Provider failed',e.message,'error')}throw Error(errs.join(' | '))}
