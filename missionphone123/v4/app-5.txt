function renderWorkspace(){
  U.runtimeButtons.forEach(b=>b.classList.toggle('active',b.dataset.runtime===S.runtime));
  U.directWorkspace.classList.toggle('hidden',S.runtime!=='direct');
  U.foundryWorkspace.classList.toggle('hidden',S.runtime!=='foundry');
  U.openRouterSummary.textContent=S.or?`${S.model.split('/').pop()} · connected`:'Sign in once';
  U.openRouterStatus.textContent=S.or?'Connected':'Connect';
  U.openRouterStatus.classList.toggle('connected',!!S.or);
  U.githubSummary.textContent=S.gh?`${S.repos.length||'…'} repositories available`:'Temporary token bridge';
  U.githubStatus.textContent=S.gh?'Connected':'Connect';
  U.githubStatus.classList.toggle('connected',!!S.gh);
  U.foundrySummary.textContent=S.foundryState?`${S.foundryState.projects?.length||0} projects · ${S.foundryState.agents?.length||0} agents`:(S.foundryBase?'Reconnect required':'Desktop or VPS runtime');
  U.foundryStatus.textContent=S.foundryState?'Connected':'Connect';
  U.foundryStatus.classList.toggle('connected',!!S.foundryState);
  if(S.gh&&!S.repos.length)loadRepos().then(renderWorkspace).catch(()=>{});
  U.startDirect.disabled=!(S.or&&S.repo&&S.branch);
  renderFoundrySelectors();
}

function renderFoundrySelectors(){
  const projects=S.foundryState?.projects||[];
  if(!projects.length){U.foundryProject.disabled=true;U.foundryAgent.disabled=true;U.startFoundry.disabled=true;return}
  const previous=U.foundryProject.value||S.activeFoundryProject?.id||S.foundryState.activeProjectId||'';
  U.foundryProject.innerHTML='';for(const p of projects){const o=document.createElement('option');o.value=p.id;o.textContent=p.name||'Project';U.foundryProject.append(o)}
  U.foundryProject.value=projects.some(p=>p.id===previous)?previous:projects[0].id;U.foundryProject.disabled=false;fillFoundryAgents();
}
function fillFoundryAgents(){const pid=U.foundryProject.value;const agents=foundryAgentsForProject(pid);const previous=U.foundryAgent.value||S.activeFoundryAgent?.id||'';U.foundryAgent.innerHTML='';for(const a of agents){const o=document.createElement('option');o.value=a.id;o.textContent=(a.name||a.role||'Agent')+(a.engine?' · '+a.engine:'');U.foundryAgent.append(o)}if(agents.length){U.foundryAgent.value=agents.some(a=>a.id===previous)?previous:agents[0].id;U.foundryAgent.disabled=false;U.startFoundry.disabled=false}else{U.foundryAgent.innerHTML='<option>No agents in this project</option>';U.foundryAgent.disabled=true;U.startFoundry.disabled=true}}
U.foundryProject.onchange=fillFoundryAgents;
