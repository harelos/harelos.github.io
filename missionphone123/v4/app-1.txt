function renderText(container,text){
  const urlRe=/(https?:\/\/[^\s]+)/g;let last=0,m;while((m=urlRe.exec(text))){if(m.index>last)container.append(document.createTextNode(text.slice(last,m.index)));let raw=m[0],trail='';while(/[),.;!?]$/.test(raw)){trail=raw.slice(-1)+trail;raw=raw.slice(0,-1)}try{const u=new URL(raw),a=document.createElement('a');a.href=u.href;a.target='_blank';a.rel='noopener noreferrer';a.textContent=raw;container.append(a)}catch{container.append(document.createTextNode(raw))}if(trail)container.append(document.createTextNode(trail));last=urlRe.lastIndex}if(last<text.length)container.append(document.createTextNode(text.slice(last)));
}
function renderAssistantContent(container,text){
  const fence=/```([a-zA-Z0-9_+-]*)\n?([\s\S]*?)```/g;let last=0,m;
  while((m=fence.exec(text))){if(m.index>last)renderText(container,text.slice(last,m.index));const lang=(m[1]||'code').toLowerCase(),code=m[2];const card=document.createElement('div');card.className='codeCard';const head=document.createElement('div');head.className='codeHead';const label=document.createElement('span');label.textContent=lang;const copy=document.createElement('button');copy.textContent='Copy';copy.onclick=async()=>{try{await navigator.clipboard.writeText(code);toast('Copied')}catch{toast('Copy unavailable')}};head.append(label,copy);const pre=document.createElement('pre');pre.textContent=code;card.append(head,pre);if(lang==='html'){const preview=document.createElement('button');preview.className='previewButton';preview.textContent='Preview';preview.onclick=()=>{const wrap=document.createElement('div');wrap.className='htmlPreview';const frame=document.createElement('iframe');frame.setAttribute('sandbox','');frame.referrerPolicy='no-referrer';frame.srcdoc=code;wrap.append(frame);card.append(wrap);preview.remove()};head.insertBefore(preview,copy)}container.append(card);last=fence.lastIndex}
  if(last<text.length)renderText(container,text.slice(last));
  const image=text.match(/https?:\/\/[^\s)]+\.(?:png|jpe?g|webp|gif)(?:\?[^\s)]*)?/i);if(image){const img=document.createElement('img');img.src=image[0];img.alt='Preview';img.loading='lazy';img.referrerPolicy='no-referrer';img.className='inlineMedia';container.append(img)}
  const video=text.match(/https?:\/\/[^\s)]+\.(?:mp4|webm|mov)(?:\?[^\s)]*)?/i);if(video){const v=document.createElement('video');v.src=video[0];v.controls=true;v.playsInline=true;v.preload='metadata';v.className='inlineMedia';container.append(v)}
}

}
